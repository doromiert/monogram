cmake_minimum_required(VERSION 3.22)
project(monogram_wallet LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
if(NOT EXISTS "${MONERO_SDK_DIR}/libmonfluo.so")
    message(FATAL_ERROR "Run scripts/prepare-monero-wallet.py before building the local wallet adapter")
endif()
add_library(monero_wallet_sdk SHARED IMPORTED)
set_target_properties(monero_wallet_sdk PROPERTIES IMPORTED_LOCATION "${MONERO_SDK_DIR}/libmonfluo.so")
add_library(monogram_wallet SHARED wallet_jni.cpp)
target_include_directories(monogram_wallet PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../third_party/monero-wallet-sdk")
target_link_libraries(monogram_wallet PRIVATE monero_wallet_sdk log)
target_link_options(monogram_wallet PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384" "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
